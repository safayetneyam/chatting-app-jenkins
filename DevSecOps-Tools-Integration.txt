# Install Betterleaks via Brew package manager

# To install Semgrep with Brew

brew install semgrep



# Download trivy, templating, and configuring | Trivy
# Source - https://stackoverflow.com/a/78982130
# Posted by nealray, modified by community. See post 'Timeline' for change history
# Retrieved 2026-09-25, License - CC BY-SA 4.0

brew install trivy

wget https://raw.githubusercontent.com/aquasecurity/trivy/main/contrib/html.tpl
sudo cp html.tpl /usr/bin/html.tpl

trivy image --format template --template "@/usr/bin/html.tpl" -o report.html {image}
